-- Chore Buddies · migration 002 · one family account on many phones
-- What this gives you:
--   * Parents sign up with email + password. Supabase emails a confirmation link,
--     and nobody can sign in until that link is tapped (so no one can use your email).
--   * The same account works on every phone and tablet: both parents and each kid's device.
--   * A second parent joins with their OWN email and a 6-letter family code.
--   * Changes sync live between devices (Supabase Realtime).
--
-- SETUP (about 10 minutes, free tier is enough)
--   1. Create a project at supabase.com.
--   2. SQL Editor → paste this whole file → Run.
--   3. Authentication → Providers → Email: keep "Confirm email" ON.
--   4. Authentication → URL Configuration: set Site URL to where the app lives,
--      e.g. https://chore-buddies.pages.dev (and add it to Redirect URLs).
--   5. Project Settings → API: copy the Project URL and the anon public key.
--   6. Host this repository on Cloudflare Pages (see SETUP.md),
--      open it, then Parent → Controls → "Use on several phones" → paste URL + key → Connect.
--      (Or paste them into SUPA_URL / SUPA_KEY at the top of the cloud section in index.html.)
--   Cloud accounts do not work inside the Claude preview, because it blocks network access.

create extension if not exists pgcrypto;

create table if not exists public.family_docs (
  id uuid primary key default gen_random_uuid(),
  data jsonb not null,                       -- the whole family: kids, tasks, worlds, chat, settings
  updated_at timestamptz not null default now(),
  updated_by uuid references auth.users(id)
);

create table if not exists public.family_members (
  family_id uuid not null references public.family_docs(id) on delete cascade,
  user_id uuid not null references auth.users(id) on delete cascade,
  role text not null default 'parent' check (role in ('owner','parent')),
  created_at timestamptz not null default now(),
  primary key (family_id, user_id)
);

create table if not exists public.family_invites (
  code text primary key,
  family_id uuid not null references public.family_docs(id) on delete cascade,
  created_by uuid default auth.uid(),
  expires_at timestamptz not null default now() + interval '7 days'
);

alter table public.family_docs enable row level security;
alter table public.family_members enable row level security;
alter table public.family_invites enable row level security;

create or replace function public.is_family_member(f uuid) returns boolean
language sql stable security definer set search_path = public as $$
  select exists (select 1 from family_members where family_id = f and user_id = auth.uid());
$$;

drop policy if exists "members read family" on public.family_docs;
drop policy if exists "members update family" on public.family_docs;
create policy "members read family" on public.family_docs for select using (is_family_member(id));
create policy "members update family" on public.family_docs for update using (is_family_member(id)) with check (is_family_member(id));

drop policy if exists "see own memberships" on public.family_members;
create policy "see own memberships" on public.family_members for select using (user_id = auth.uid() or is_family_member(family_id));

drop policy if exists "members create invites" on public.family_invites;
drop policy if exists "members see invites" on public.family_invites;
create policy "members create invites" on public.family_invites for insert with check (is_family_member(family_id));
create policy "members see invites" on public.family_invites for select using (is_family_member(family_id));

-- New family for a freshly confirmed parent.
create or replace function public.create_family(p_data jsonb) returns uuid
language plpgsql security definer set search_path = public as $$
declare f uuid;
begin
  if auth.uid() is null then raise exception 'Please sign in first'; end if;
  insert into family_docs (data, updated_by) values (p_data, auth.uid()) returning id into f;
  insert into family_members (family_id, user_id, role) values (f, auth.uid(), 'owner');
  return f;
end $$;

-- Second parent joins with a code.
create or replace function public.join_family(p_code text) returns uuid
language plpgsql security definer set search_path = public as $$
declare f uuid;
begin
  if auth.uid() is null then raise exception 'Please sign in first'; end if;
  select family_id into f from family_invites where code = upper(p_code) and expires_at > now();
  if f is null then raise exception 'That family code is not valid or has expired'; end if;
  insert into family_members (family_id, user_id, role) values (f, auth.uid(), 'parent') on conflict do nothing;
  return f;
end $$;

grant execute on function public.create_family(jsonb) to authenticated;
grant execute on function public.join_family(text) to authenticated;

-- Live updates between phones.
alter publication supabase_realtime add table public.family_docs;
